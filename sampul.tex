% Sampul depan identik dengan halaman pertama main.pdf. Punggung: nama + NRP
% (12 pt) dan judul (14 pt), tebal, putih, ditulis vertikal; warna RGB

\documentclass[digital,nocover]{pens-thesis}
\input{config/metadata}

% Lebar punggung buku: tanyakan ke percetakan (tergantung jumlah halaman)
\newlength{\punggung}
\setlength{\punggung}{20mm}

\newlength{\lebarsampul}
\setlength{\lebarsampul}{\dimexpr 420mm + \punggung\relax}
\geometry{paperwidth=\lebarsampul,paperheight=297mm,margin=0mm}
\pagestyle{empty}

\begin{document}
\makeatletter
\noindent
\begin{tikzpicture}[remember picture,overlay]
  \coordinate (NW) at (current page.north west);
  % Sampul belakang + punggung
  \fill[pensblue] (NW) rectangle ([xshift=210mm+\punggung,yshift=-297mm]NW);
  % Sampul depan
  \pens@drawfrontcover{[xshift=210mm+\punggung]NW}
  % Punggung: teks vertikal, dibaca dari atas ke bawah
  \begin{scope}[shift={([xshift=210mm+0.5\punggung]NW)}]
    \draw[white,very thin,dashed] (-0.5\punggung,0) -- (-0.5\punggung,-297mm);
    \draw[white,very thin,dashed] (0.5\punggung,0) -- (0.5\punggung,-297mm);
    \node[rotate=-90,text=white,align=center,
          font=\pens@fs{12}\bfseries] at (0,-40mm)
      {\pens@penulis\\NRP.~\pens@nrp};
    \node[rotate=-90,text=white,text width=170mm,align=center,
          font=\pens@fs{14}\bfseries] at (0,-175mm)
      {\MakeUppercase{\pens@judul}};
  \end{scope}
\end{tikzpicture}
\makeatother
\null
\end{document}
